cmake_minimum_required(VERSION 3.21)
project(VitaMTPConsumer C)
find_package(VitaMTP 2.5.12 CONFIG REQUIRED)
add_executable(vitamtp_consumer main.c)
option(CONSUME_STATIC "Link the installed static library" OFF)
if(CONSUME_STATIC)
  target_link_libraries(vitamtp_consumer PRIVATE VitaMTP::VitaMTPStatic)
else()
  target_link_libraries(vitamtp_consumer PRIVATE VitaMTP::VitaMTP)
endif()
enable_testing()
add_test(NAME consumer COMMAND vitamtp_consumer)
